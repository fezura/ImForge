file(GLOB IMFORGE_TEST_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*_tests.cpp")

if(NOT IMFORGE_TEST_SOURCES)
    message(STATUS "ImForge: no tests/*_tests.cpp yet")
endif()

# libs are picked from the file name (tests/<name>_tests.cpp) or its includes, so adding a test
# file normally doesn't need an edit here
foreach(_src IN LISTS IMFORGE_TEST_SOURCES)
    get_filename_component(_file "${_src}" NAME_WE)
    string(REGEX REPLACE "_tests$" "" _name "${_file}")
    file(STRINGS "${_src}" _inc REGEX "^[ \t]*#[ \t]*include")
    set(_extra)
    set(_builder_inc OFF)
    if(_name MATCHES "^(codegen|integration)" OR _inc MATCHES "\"(codegen|emit_ctx|naming|types_db)\\.h\"")
        list(APPEND _extra imforge_codegen)
    endif()
    if(_name MATCHES "^(roundtrip|integration)" OR _inc MATCHES "\"roundtrip\\.h\"")
        list(APPEND _extra imforge_roundtrip imforge_codegen)
    endif()
    if(_name MATCHES "^(binding|integration)" OR _inc MATCHES "\"(clang_scanner|compile_commands|panel_type_env|reflect_header|type_database)\\.h\"")
        list(APPEND _extra imforge_binding)
    endif()
    if(_name MATCHES "^(essentials|integration)" OR _inc MATCHES "imb_ess")
        list(APPEND _extra imforge_essentials)
    endif()
    if(_name MATCHES "^(imblib|pack|integration)" OR _inc MATCHES "imblib|miniz")
        list(APPEND _extra miniz)
        set(_builder_inc ON)
    endif()
    if(_name MATCHES "^(tree_ops|builder)" OR _inc MATCHES "src/builder")
        set(_builder_inc ON)
    endif()
    if(_name MATCHES "(transport|protocol|client|live|integration)" OR _inc MATCHES "imforge_transport|imforge_client|imforge_protocol")
        if(WIN32)
            list(APPEND _extra ws2_32)
        elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
            list(APPEND _extra rt)
        endif()
    endif()
    list(REMOVE_DUPLICATES _extra)
    imforge_add_test(${_name} ${_extra})
    if(_builder_inc AND TARGET ${_name}_tests)
        target_include_directories(${_name}_tests PRIVATE "${PROJECT_SOURCE_DIR}/src/builder")
    endif()
    if(IMFORGE_HAVE_LIBCLANG AND IMFORGE_LIBCLANG_DLL AND "imforge_binding" IN_LIST _extra AND TARGET ${_name}_tests)
        add_custom_command(TARGET ${_name}_tests POST_BUILD
            COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${IMFORGE_LIBCLANG_DLL}" "$<TARGET_FILE_DIR:${_name}_tests>"
            VERBATIM)
    endif()
    if(_name MATCHES "^integration" AND TARGET ${_name}_tests)
        set_tests_properties(${_name}_tests PROPERTIES TIMEOUT 1800)
    endif()
endforeach()
