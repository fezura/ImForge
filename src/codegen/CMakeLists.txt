file(GLOB IMFORGE_CODEGEN_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp")

if(IMFORGE_CODEGEN_SOURCES)
    add_library(imforge_codegen STATIC ${IMFORGE_CODEGEN_SOURCES})
    target_include_directories(imforge_codegen PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
    target_link_libraries(imforge_codegen PUBLIC imforge::headers imgui PRIVATE imforge::warnings)
else()
    message(STATUS "ImForge: no sources in src/codegen yet, imforge_codegen is an INTERFACE placeholder")
    add_library(imforge_codegen INTERFACE)
    target_include_directories(imforge_codegen INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_link_libraries(imforge_codegen INTERFACE imforge::headers imgui)
endif()
add_library(imforge::codegen ALIAS imforge_codegen)
set_target_properties(imforge_codegen PROPERTIES FOLDER "libs")
