# libclang lookup, IMFORGE_WITH_LIBCLANG is AUTO, ON or OFF
set(IMFORGE_HAVE_LIBCLANG OFF)
if(NOT IMFORGE_WITH_LIBCLANG STREQUAL "OFF")
    set(_llvm_hints)
    if(LLVM_DIR)
        list(APPEND _llvm_hints "${LLVM_DIR}" "${LLVM_DIR}/../../..")
    endif()
    if(DEFINED ENV{LLVM_DIR})
        list(APPEND _llvm_hints "$ENV{LLVM_DIR}" "$ENV{LLVM_DIR}/../../..")
    endif()
    if(DEFINED ENV{LLVM_ROOT})
        list(APPEND _llvm_hints "$ENV{LLVM_ROOT}")
    endif()
    if(WIN32)
        list(APPEND _llvm_hints "C:/Program Files/LLVM" "$ENV{ProgramFiles}/LLVM")
    else()
        file(GLOB _llvm_versioned "/usr/lib/llvm-*" "/usr/local/opt/llvm" "/opt/homebrew/opt/llvm" "/usr/local/llvm*")
        list(SORT _llvm_versioned COMPARE NATURAL ORDER DESCENDING)
        list(APPEND _llvm_hints ${_llvm_versioned} /usr /usr/local)
    endif()
    find_path(IMFORGE_LIBCLANG_INCLUDE_DIR clang-c/Index.h HINTS ${_llvm_hints} PATH_SUFFIXES include)
    find_library(IMFORGE_LIBCLANG_LIBRARY
        NAMES clang libclang clang-20 clang-19 clang-18 clang-17 clang-16 clang-15 clang-14
        HINTS ${_llvm_hints} PATH_SUFFIXES lib lib64 bin)
    if(WIN32)
        find_file(IMFORGE_LIBCLANG_DLL libclang.dll HINTS ${_llvm_hints} PATH_SUFFIXES bin)
    endif()
    if(IMFORGE_LIBCLANG_INCLUDE_DIR AND IMFORGE_LIBCLANG_LIBRARY)
        set(IMFORGE_HAVE_LIBCLANG ON)
        message(STATUS "ImForge: libclang found: ${IMFORGE_LIBCLANG_LIBRARY} (${IMFORGE_LIBCLANG_INCLUDE_DIR})")
    elseif(IMFORGE_WITH_LIBCLANG STREQUAL "ON")
        message(FATAL_ERROR "IMFORGE_WITH_LIBCLANG=ON but libclang was not found (set LLVM_DIR to the LLVM install root)")
    else()
        message(STATUS "ImForge: libclang not found, imforge_binding is built without it")
    endif()
endif()
set(IMFORGE_HAVE_LIBCLANG ${IMFORGE_HAVE_LIBCLANG} PARENT_SCOPE)

file(GLOB IMFORGE_BINDING_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp")
list(FILTER IMFORGE_BINDING_SOURCES EXCLUDE REGEX "reflect_main\\.cpp$")

if(IMFORGE_BINDING_SOURCES)
    add_library(imforge_binding STATIC ${IMFORGE_BINDING_SOURCES})
    target_include_directories(imforge_binding PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
    target_link_libraries(imforge_binding PUBLIC imforge::headers imgui PRIVATE imforge::warnings)
    if(IMFORGE_HAVE_LIBCLANG)
        target_compile_definitions(imforge_binding PUBLIC IMFORGE_HAVE_LIBCLANG=1)
        target_include_directories(imforge_binding PUBLIC "${IMFORGE_LIBCLANG_INCLUDE_DIR}")
        target_link_libraries(imforge_binding PUBLIC "${IMFORGE_LIBCLANG_LIBRARY}")
    endif()
else()
    message(STATUS "ImForge: no sources in src/binding yet, imforge_binding is an INTERFACE placeholder")
    add_library(imforge_binding INTERFACE)
    target_include_directories(imforge_binding INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_link_libraries(imforge_binding INTERFACE imforge::headers imgui)
    if(IMFORGE_HAVE_LIBCLANG)
        target_compile_definitions(imforge_binding INTERFACE IMFORGE_HAVE_LIBCLANG=1)
        target_include_directories(imforge_binding INTERFACE "${IMFORGE_LIBCLANG_INCLUDE_DIR}")
        target_link_libraries(imforge_binding INTERFACE "${IMFORGE_LIBCLANG_LIBRARY}")
    endif()
endif()
add_library(imforge::binding ALIAS imforge_binding)
set_target_properties(imforge_binding PROPERTIES FOLDER "libs")

if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/reflect_main.cpp")
    add_executable(imforge-reflect "${CMAKE_CURRENT_SOURCE_DIR}/reflect_main.cpp")
    target_link_libraries(imforge-reflect PRIVATE imforge_binding imforge::headers imgui Threads::Threads imforge::warnings)
    set_target_properties(imforge-reflect PROPERTIES FOLDER "tools")
    if(IMFORGE_HAVE_LIBCLANG AND IMFORGE_LIBCLANG_DLL)
        add_custom_command(TARGET imforge-reflect POST_BUILD
            COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${IMFORGE_LIBCLANG_DLL}" "$<TARGET_FILE_DIR:imforge-reflect>"
            VERBATIM)
    endif()
else()
    message(STATUS "ImForge: src/binding/reflect_main.cpp not present yet, imforge-reflect is skipped")
endif()
