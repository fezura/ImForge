file(GLOB IMFORGE_ROUNDTRIP_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp")

if(IMFORGE_ROUNDTRIP_SOURCES)
    add_library(imforge_roundtrip STATIC ${IMFORGE_ROUNDTRIP_SOURCES})
    target_include_directories(imforge_roundtrip PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
    target_link_libraries(imforge_roundtrip PUBLIC imforge::headers imgui imforge_codegen PRIVATE imforge::warnings)
else()
    message(STATUS "ImForge: no sources in src/roundtrip yet, imforge_roundtrip is an INTERFACE placeholder")
    add_library(imforge_roundtrip INTERFACE)
    target_include_directories(imforge_roundtrip INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_link_libraries(imforge_roundtrip INTERFACE imforge::headers imgui imforge_codegen)
endif()
add_library(imforge::roundtrip ALIAS imforge_roundtrip)
set_target_properties(imforge_roundtrip PROPERTIES FOLDER "libs")
